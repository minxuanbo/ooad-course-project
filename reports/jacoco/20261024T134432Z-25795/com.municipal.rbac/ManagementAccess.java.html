package com.municipal.rbac;

import java.util.*;
import org.springframework.jdbc.core.JdbcTemplate;

/** Each grant binds one operation to one scope; scopes never cross-multiply permissions. */
class ManagementAccess {
  static final Map<String,String> PERMISSIONS = Collections.unmodifiableMap(new LinkedHashMap<>(Map.ofEntries(
    Map.entry("user.read","人员查询"), Map.entry("user.roles","用户角色分配"),
    Map.entry("role.read","角色查看"), Map.entry("role.permissions","角色权限配置"),
    Map.entry("org.read","组织查看"), Map.entry("org.write","组织维护"),
    Map.entry("app.read","系统目录查看"), Map.entry("audit.read","审计日志查看"),
    Map.entry("handover.start","离职发起"), Map.entry("handover.complete","离职完成"),
    Map.entry("management.grant","管理授权配置"))));
  private final JdbcTemplate db;
  ManagementAccess(JdbcTemplate db){this.db=db;}
  List<Map<String,Object>> grants(long uid){
    return db.queryForList("SELECT g.id,g.permission_code permission,g.scope_type scope,g.company_id companyId,g.department_id departmentId,g.include_descendants descendants,c.name company,d.name department FROM management_grants g JOIN users u ON u.id=g.user_id LEFT JOIN companies c ON c.id=g.company_id LEFT JOIN departments d ON d.id=g.department_id WHERE g.user_id=? AND u.status='ACTIVE' ORDER BY g.id",uid);
  }
  boolean has(long uid,String permission){
    return db.queryForObject("SELECT count(*) FROM management_grants g JOIN users u ON u.id=g.user_id WHERE g.user_id=? AND g.permission_code=? AND u.status='ACTIVE'",Integer.class,uid,permission)>0;
  }
  boolean global(long uid,String permission){
    return db.queryForObject("SELECT count(*) FROM management_grants g JOIN users u ON u.id=g.user_id WHERE g.user_id=? AND g.permission_code=? AND g.scope_type='GLOBAL' AND u.status='ACTIVE'",Integer.class,uid,permission)>0;
  }
  void require(long uid,String permission){if(!has(uid,permission))throw ApiException.forbidden();}
  void requireGlobal(long uid,String permission){if(!global(uid,permission))throw ApiException.forbidden();}
  // uid and SQL expressions are server-owned, permission is always a catalog constant.
  String scope(long uid,String permission,String department){
    if(!PERMISSIONS.containsKey(permission))throw new IllegalArgumentException("Unknown management permission");
    return "EXISTS(SELECT 1 FROM management_grants g JOIN users actor ON actor.id=g.user_id WHERE g.user_id="+uid+
      " AND actor.status='ACTIVE' AND g.permission_code='"+permission+"' AND (g.scope_type='GLOBAL' OR (g.scope_type='COMPANY' AND g.company_id=(SELECT sd.company_id FROM departments sd WHERE sd.id="+department+")) OR (g.scope_type='DEPARTMENT' AND (g.department_id="+department+
      " OR (g.include_descendants=1 AND "+department+" IN (WITH RECURSIVE descendants AS (SELECT id FROM departments WHERE id=g.department_id UNION ALL SELECT child.id FROM departments child JOIN descendants parent ON child.parent_id=parent.id) SELECT id FROM descendants))))))";
  }
  void user(long uid,String permission,long target){
    require(uid,permission);
    if(global(uid,permission))return;
    if(db.queryForObject("SELECT count(*) FROM users u WHERE u.id=? AND "+scope(uid,permission,"u.department_id"),Integer.class,target)==0)throw ApiException.forbidden();
  }
  void department(long uid,String permission,long target){
    if(db.queryForObject("SELECT count(*) FROM departments d WHERE d.id=? AND "+scope(uid,permission,"d.id"),Integer.class,target)==0)throw ApiException.forbidden();
  }
  void lock(){db.queryForObject("SELECT version FROM management_migrations WHERE version=1 FOR UPDATE",Integer.class);}
  void protectTarget(long actor,String permission,long target){
    user(actor,permission,target);
    if(!global(actor,permission) && (actor==target || db.queryForObject("SELECT count(*) FROM management_grants WHERE user_id=?",Integer.class,target)>0))throw ApiException.forbidden();
  }
  void retainPlatformAdmin(){
    if(db.queryForObject("SELECT count(*) FROM management_grants g JOIN users u ON u.id=g.user_id WHERE g.permission_code='management.grant' AND g.scope_type='GLOBAL' AND u.status='ACTIVE'",Integer.class)==0)
      throw ApiException.bad("必须保留至少一名可配置管理授权的平台管理员");
  }
}
