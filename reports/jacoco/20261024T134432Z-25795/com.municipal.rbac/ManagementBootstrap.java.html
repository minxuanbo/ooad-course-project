package com.municipal.rbac;

import org.springframework.boot.ApplicationArguments;
import org.springframework.boot.ApplicationRunner;
import org.springframework.core.annotation.Order;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.Transactional;

/** One-time compatibility migration, after Excel initialization. Never regrants on restart. */
@Component @Order(20)
class ManagementBootstrap implements ApplicationRunner {
  private final JdbcTemplate db;
  ManagementBootstrap(JdbcTemplate db){this.db=db;}
  @Transactional public void run(ApplicationArguments args){
    db.update("INSERT IGNORE INTO management_migrations(version,completed) VALUES(1,0)");
    if(db.queryForObject("SELECT completed FROM management_migrations WHERE version=1 FOR UPDATE",Integer.class)==1)return;
    for(String permission:ManagementAccess.PERMISSIONS.keySet())
      db.update("INSERT INTO management_grants(user_id,permission_code,scope_type) SELECT DISTINCT u.id,?,'GLOBAL' FROM users u JOIN user_roles ur ON ur.user_id=u.id JOIN roles r ON r.id=ur.role_id WHERE u.status='ACTIVE' AND r.code='系统管理员' AND r.enabled=1",permission);
    new ManagementAccess(db).retainPlatformAdmin();
    db.update("UPDATE management_migrations SET completed=1 WHERE version=1");
  }
}
